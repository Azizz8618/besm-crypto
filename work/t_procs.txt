(* --- имя области по-новому (УПП): 'ТЕСТ/419999', 0-терминатор --- *)
(* Т=62(8)=50 Е=45(8)=37 С=61(8)=49 Т=50 /=12 4=4 1=1 9=9 (ГОСТ) *)
procedure setname;
begin
  isw[2].l := 50*65536 + 37*256 + 49;    (* Т Е С *)
  isw[2].r := 50*65536 + 12*256 + 4;     (* Т / 4 *)
  isw[3].l := 1*65536 + 9*256 + 9;       (* 1 9 9 *)
  isw[3].r := 9*65536 + 9*256            (* 9 9 0 *)
end;
(* --- Э63 01: создать область ТЕСТ/419999 (ln зон) --- *)
procedure mkarea(ln: integer);
begin

  isw[1].l := 1000000B;      (* тип 01 - создание области *)
  isw[1].r := 8;             (* имя по-новому (10В) *)

(* --- печать хеша/массива слов (8 слов по 4 в строке) --- *)
procedure prhash(var h: Array [1..8] of UInt32);
begin
  for ii := 1 to 8 do begin
    write(h[ii]:11);
    if ii = 4 then writeln
  end;
  writeln
end;
(* --- печать шифртекста по 4 слова в строке --- *)
procedure prwords(var a: Array[1..101] of Integer; n: Integer);
var t: Integer;
begin
  for t := 1 to n do begin
    write(a[t]:11);
    if t mod 4 = 0 then writeln
  end;
  if n mod 4 <> 0 then writeln
end;

(* --- гаммирование сообщения (ГОСТ 28147-89, счётчик): --- *)
(* коды символов -> слова шифртекста; dst[1] = синхропосылка IV, *)
(* далее по слову на символ (по образцу EncryptToArray golden)    *)
procedure EncChars(
  ctx: TypeGostCtx;
  iv: Integer;
  var src: Array[1..100] of Integer;
  len: Integer;
  var dst: Array[1..101] of Integer);
var
  counter, gamma: Array[1..2] of UInt32;
  t: Integer;
begin
  dst[1] := iv;
  counter[1] := iv;
  counter[2] := 0;
  t := 0;
  while t < len do
  begin
    counter[2] := counter[2] + 1;
    gamma[1] := counter[1];
    gamma[2] := counter[2];
    EncryptBlock(ctx, gamma);
    t := t + 1;
    dst[t+1] := Xor(src[t], gamma[1]);
    if t < len then
    begin
      t := t + 1;
      dst[t+1] := Xor(src[t], gamma[2])
    end
  end
end;
(* --- обратное гаммирование: [1]=IV, далее шифртекст -> коды --- *)
procedure DecChars(
  ctx: TypeGostCtx;
  var arr: Array[1..101] of Integer;
  len: Integer;
  var res: Array[1..100] of Integer);
var
  counter, gamma: Array[1..2] of UInt32;
  t: Integer;
begin
  counter[1] := arr[1];
  counter[2] := 0;
  t := 0;
  while t < len do
  begin
    counter[2] := counter[2] + 1;
    gamma[1] := counter[1];
    gamma[2] := counter[2];
    EncryptBlock(ctx, gamma);
    t := t + 1;
    res[t] := Xor(arr[t+1], gamma[1]);
    if t < len then
    begin
      t := t + 1;
      res[t] := Xor(arr[t+1], gamma[2])
    end
  end
end;
